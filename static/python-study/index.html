预期输出
[3, 6, 6]
参考写法
from collections import deque

values = [3, 2, 6, 5]
k = 2
q = deque()
out = []
for i, value in enumerate(values):
    while q and q[0] <= i - k:
        q.popleft()

        q.pop()
    q.append(i)
    if i >= k - 1:
        out.append(values[q[0]])
